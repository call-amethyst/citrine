\chapter{Reihen}\label{chap:Reihen} % (fold)

\section{Reihen}

Eine besonders wichtige Klasse von Folgen sind die sogenannten
\textbf{Reihen}. Die Untersuchung von Reihen und deren Verhalten
ist eine zentrale Aufgabe der Analysis.

Reihen werden aus Folgen konstruiert.

\dfn{Reihen}{
  Sei $a$ eine Folge reeller Zahlen. Dann heißt die Folge
  $S$ mit Folgengliedern
  \[
    S_n:=\sum_{k=1}^n a_k
  \]
  eine \textbf{Reihe}.

  Wenn $S$ gegen eine reelle Zahl konvergiert, heißt
  die Reihe \textbf{konvergent}. Man schreibt
  \[
    \lim_{n\to\infty}S_n
    =\lim_{n\to\infty}\sum_{k=1}^n a_k
    =\sum_{k=1}^{\infty}a_k.
  \]
}

\nt{
  Die Schreibweise
  \[
    \sum_{k=1}^{\infty}a_k
  \]
  ist durch den Grenzwert der Partialsummen definiert.

  Äquivalent schreibt man auch
  \[
    \sum_{k\in\bbN}a_k.
  \]

  Reihen können ebenso bei $k=0$ oder einem anderen
  Index beginnen.

  Die Zahlen $a_k$ heißen \textbf{Reihenglieder},
  die Zahlen
  \[
    S_n=\sum_{k=1}^n a_k
  \]
  heißen \textbf{Partialsummen} der Reihe.

  Häufig bezeichnet man auch die gesamte Reihe mit
  $\sum_{k=1}^{\infty}a_k$, obwohl diese Schreibweise
  streng genommen ihren Grenzwert bezeichnet.
}


\subsection{Die geometrische Reihe}

\dfn{Geometrische Reihe}{
  Sei $x\in\bbR$. Die Reihe $S$ mit
  \[
    S_n:=\sum_{k=1}^n x^{k-1}
        =\sum_{k=0}^{n-1}x^k
  \]
  heißt \textbf{geometrische Reihe}.
}

\mlemma{Konvergenz der geometrischen Reihe}{
  Sei $S$ die geometrische Reihe. Für $x\ne1$ gilt
  \[
    \boxed{
      S_n=\frac{1-x^n}{1-x}.
    }
  \]

  Für $x=1$ gilt
  \[
    S_n=n.
  \]

  Die geometrische Reihe konvergiert genau dann,
  wenn $|x|<1$. In diesem Fall gilt
  \[
    \boxed{
      \sum_{k=0}^{\infty}x^k=\frac{1}{1-x}.
    }
  \]
}

\begin{myproof}
  Die Formel für die Partialsummen
  \[
    S_n=\frac{1-x^n}{1-x}
  \]
  wurde bereits mithilfe der geometrischen Summenformel
  hergeleitet.

  \textbf{Fall 1:} $|x|<1$.

  Es gilt
  \[
    S_n-\frac{1}{1-x}
    =-\frac{x^n}{1-x}.
  \]

  Da $x^n\to0$ für $|x|<1$, folgt
  \[
    \lim_{n\to\infty}S_n=\frac{1}{1-x}.
  \]

  Somit konvergiert die geometrische Reihe.

  \medskip
  \textbf{Fall 2:} $|x|\ge1$.

  Für alle $n\in\bbN$ gilt
  \[
    |S_{n+1}-S_n|=|x|^n\ge1.
  \]

  Daher ist $(S_n)$ keine Cauchy-Folge und kann
  somit nicht konvergieren.

  Insgesamt konvergiert die geometrische Reihe
  genau dann, wenn $|x|<1$.
\end{myproof}


\subsection{Konvergenz von Reihen}

Bei der geometrischen Reihe haben wir gesehen, dass
die Reihe genau dann konvergiert, wenn ihre
Reihenglieder gegen $0$ konvergieren.

Es stellt sich die Frage, ob dies allgemein gilt.

\mlemma{Notwendige Bedingung für Reihenkonvergenz}{
  Sei $a$ eine reelle Folge und $S$ die Reihe mit
  Partialsummen
  \[
    S_n=\sum_{k=1}^n a_k.
  \]

  Wenn $S$ konvergiert, dann konvergiert auch
  die Folge $a$, und es gilt
  \[
    \boxed{\lim_{n\to\infty}a_n=0.}
  \]
}

\begin{myproof}
  Da $S$ konvergiert, ist $(S_n)$ eine Cauchy-Folge.

  Somit existiert für jedes $\varepsilon>0$
  ein $n_0\in\bbN$, sodass für alle $n\ge n_0$ gilt:
  \[
    |S_{n+1}-S_n|<\varepsilon.
  \]

  Wegen
  \[
    S_{n+1}-S_n=a_{n+1}
  \]
  folgt
  \[
    |a_{n+1}|<\varepsilon.
  \]

  Also konvergiert die Folge der Reihenglieder
  gegen $0$:
  \[
    \lim_{n\to\infty}a_n=0.
  \]
\end{myproof}

\nt{
  Die Bedingung
  \[
    \lim_{n\to\infty}a_n=0
  \]
  ist \textbf{notwendig}, aber im Allgemeinen
  \textbf{nicht hinreichend} für die Konvergenz von
  \[
    \sum_{k=1}^{\infty}a_k.
  \]

  Die Umkehrung des vorherigen Lemmas gilt also nicht.
}
