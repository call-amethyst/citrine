\chapter{Algorithmen}

\section{Motivation}

\dfn{Elementare Addition}{
  Die elementare Addition bezeichnet die Addition dreier einstelliger
  Zahlen, deren Ergebnis höchstens zweistellig ist.
}

\nt{
  Zur Addition zweier $n$-stelliger Zahlen benötigt man im üblichen
  schriftlichen Additionsverfahren $\mathcal{O}(n)$ elementare
  Rechenschritte.

  Für die Multiplikation zweier $n$-stelliger Zahlen ist die Situation
  wesentlich komplexer. Die Laufzeit hängt vom verwendeten Algorithmus
  ab. Moderne Multiplikationsalgorithmen erreichen eine Laufzeit von
  $\mathcal{O}(n\log n)$ bis auf zusätzliche logarithmische Faktoren.
}

\dfn{Elementare Multiplikation}{
  Die elementare Multiplikation bezeichnet die Multiplikation zweier
  einstelliger Zahlen, deren Ergebnis höchstens zweistellig ist.
}

\nt{
  Die Frage, wie schnell sich zwei große Zahlen multiplizieren lassen,
  ist ein zentrales Problem der algorithmischen Mathematik. Es existieren
  Algorithmen, die asymptotisch wesentlich schneller sind als die
  klassische schriftliche Multiplikation.
}

\ex{Numerische Instabilität}{
  Betrachte das lineare Gleichungssystem
  \[
    \begin{aligned}
      10^{-20}x + 2y &= 1, \\
      10^{-20}x + 10^{-20}y &= 10^{-20}.
    \end{aligned}
  \]

  Subtrahiert man die zweite Gleichung von der ersten, erhält man
  \[
    (2-10^{-20})y = 1-10^{-20}.
  \]
  Damit folgt
  \[
    y = \frac{1-10^{-20}}{2-10^{-20}}
      \approx 0{,}5.
  \]

  Wird dieser gerundete Wert in die ursprünglichen Gleichungen
  eingesetzt, können aufgrund der Rundungsfehler widersprüchliche
  Ergebnisse für $x$ entstehen.

  Die exakte Lösung lautet hingegen
  \[
    \begin{aligned}
      y &= \frac{1-10^{-20}}{2-10^{-20}}, \\
      x &= \frac{1}{2-10^{-20}}.
    \end{aligned}
  \]

  Beide Werte liegen sehr nahe bei $0{,}5$. Das Beispiel illustriert,
  dass Rundungsfehler bei numerischen Berechnungen zu erheblichen
  Ungenauigkeiten führen können.
}

\section{Mengen und Relationen}

\dfn{Kartesisches Produkt}{
  Seien $A$ und $B$ Mengen. Das \textbf{kartesische Produkt} von $A$
  und $B$ ist definiert durch
  \[
    A \times B
    := \{(a,b) \mid a \in A,\ b \in B\}.
  \]

  Eine \textbf{Relation} zwischen $A$ und $B$ ist eine Teilmenge
  $R \subseteq A \times B$.

  Für $(a,b) \in R$ schreibt man auch $a\,R\,b$.
}

\ex{Kartesisches Produkt und Relation}{
  Seien
  \[
    A = \{1,3,5\},
    \qquad
    B = \{2,5\}.
  \]
  Dann ist
  \[
    A \times B
    = \{(1,2),(1,5),(3,2),(3,5),(5,2),(5,5)\}.
  \]

  Eine mögliche Relation zwischen $A$ und $B$ ist beispielsweise
  \[
    R = \{(1,2),(3,2),(5,5)\}.
  \]
  Es gilt $R \subseteq A \times B$. Die Relation enthält genau die
  ausgewählten geordneten Paare.
}

\dfn{Abbildung}{
  Seien $A$ und $B$ Mengen. Eine Teilmenge $f \subseteq A \times B$
  heißt \textbf{Funktion} oder \textbf{Abbildung} von $A$ nach $B$,
  falls für jedes $a \in A$ genau ein $b \in B$ existiert, sodass
  $(a,b) \in f$ gilt.

  Schreibweisen:
  \[
    f \colon A \to B, \qquad
    (a,b) \in f \iff f(a)=b, \qquad
    a \mapsto f(a).
  \]
  Die Menge $A$ heißt \textbf{Definitionsbereich}, die Menge $B$
  \textbf{Zielmenge} von $f$.
}

\dfn{Injektivität}{
  Eine Funktion $f \colon A \to B$ heißt \textbf{injektiv}, falls
  für alle $x,y \in A$ gilt:
  \[
    f(x)=f(y) \implies x=y.
  \]
  Äquivalent dazu gilt
  \[
    x \neq y \implies f(x) \neq f(y).
  \]
}

\dfn{Surjektivität}{
  Eine Funktion $f \colon A \to B$ heißt \textbf{surjektiv}, falls
  jedes Element der Zielmenge angenommen wird, das heißt
  \[
    \forall b \in B\ \exists a \in A: f(a)=b.
  \]
}

\dfn{Bijektivität}{
  Eine Funktion $f \colon A \to B$ heißt \textbf{bijektiv}, falls
  sie sowohl injektiv als auch surjektiv ist.
  In diesem Fall wird jedes Element von $B$ von genau einem Element
  aus $A$ angenommen.
}

\ex{Injektivität und Surjektivität}{
  Betrachte die Funktion
  \[
    f \colon \bbN \to \bbN, \qquad f(x)=x^2.
  \]
  Für $a,a' \in \bbN$ gilt
  \[
    a^2=(a')^2 \implies a=a',
  \]
  also ist $f$ injektiv.

  Die Funktion ist jedoch nicht surjektiv, da beispielsweise
  $2 \in \bbN$ kein Urbild besitzt: Es gibt kein $a \in \bbN$
  mit $a^2=2$. Folglich ist $f$ nicht bijektiv.

  Dagegen ist die Funktion
  \[
    f \colon \{0\} \to \{0\}, \qquad f(0)=0,
  \]
  bijektiv.
}

\dfn{Ganzzahlige Intervalle}{
  Für $a,b \in \bbZ$ bezeichne
  \[
    \{a,\dots,b\}
    := \{x \in \bbZ \mid a \leq x \leq b\}.
  \]
  Für $b<a$ setzen wir
  \[
    \{a,\dots,b\} := \emptyset.
  \]
}

\dfn{Endlichkeit}{
  Eine Menge $A$ heißt \textbf{endlich}, falls ein $n \in \bbN_0$
  und eine injektive Funktion
  \[
    f \colon A \to \{1,\dots,n\}
  \]
  existieren.

  Andernfalls heißt $A$ \textbf{unendlich}.
  Für eine endliche Menge $A$ bezeichnet $|A|$ die Anzahl ihrer
  Elemente.
}

\dfn{Abzählbarkeit}{
  Eine Menge $A$ heißt \textbf{abzählbar}, falls eine injektive
  Funktion $f \colon A \to \bbN$ existiert.

  Eine Menge, die nicht abzählbar ist, heißt
  \textbf{überabzählbar}.
}

\dfn{Wörter und Sprachen}{
  Sei $A$ eine endliche, nicht leere Menge, die als
  \textbf{Alphabet} bezeichnet wird. Für $k \in \bbN_0$ sei
  \[
    A^k := \{f \mid f \colon \{1,\dots,k\} \to A\}
  \]
  die Menge aller Wörter der Länge $k$ über dem Alphabet $A$.

  Ein Element $f \in A^k$ wird mit der Folge
  \[
    f(1)f(2)\cdots f(k)
  \]
  identifiziert und als \textbf{Wort der Länge $k$} bezeichnet.

  Die Menge $A^0$ enthält genau ein Element, das
  \textbf{leere Wort} $\varepsilon$.

  Die Menge aller endlichen Wörter über $A$ ist
  \[
    A^* := \bigcup_{k \in \bbN_0} A^k.
  \]
  Eine \textbf{Sprache} über dem Alphabet $A$ ist eine Teilmenge
  von $A^*$.
}

\dfn{Berechnungsproblem}{
  Ein \textbf{Berechnungsproblem} ist eine Relation
  \[
    P \subseteq D \times E,
  \]
  wobei $D$ die Menge der Eingaben und $E$ die Menge der möglichen
  Ausgaben bezeichnet. Es wird vorausgesetzt, dass zu jeder Eingabe
  $d \in D$ mindestens eine Ausgabe $e \in E$ existiert, sodass
  $(d,e) \in P$ gilt.
}

\nt{
  Ein Paar $(d,e) \in P$ beschreibt eine Lösung des Problems für
  die Eingabe $d$. In diesem Fall heißt $e$ eine
  \textbf{korrekte Ausgabe} für die Eingabe $d$.

  Die Elemente von $D$ heißen \textbf{Instanzen} des Problems.
}

\dfn{Eindeutigkeit und Problemtypen}{
  Ein Berechnungsproblem $P \subseteq D \times E$ heißt
  \textbf{eindeutig}, falls zu jeder Eingabe $d \in D$ genau eine
  korrekte Ausgabe $e \in E$ existiert. In diesem Fall beschreibt
  $P$ eine Funktion $P \colon D \to E$.

  Sind $D$ und $E$ Sprachen über einem endlichen Alphabet $A$,
  so heißt $P$ ein \textbf{diskretes Berechnungsproblem}.

  Sind $D \subseteq \bbR^m$ und $E \subseteq \bbR^n$ für
  $m,n \in \bbN$, so heißt $P$ ein \textbf{numerisches
  Berechnungsproblem}.

  Ein eindeutiges Berechnungsproblem $P \colon D \to E$ mit
  $|E|=2$ heißt \textbf{Entscheidungsproblem}.
}

\section{Algorithmen und Programme}

\dfn{Bit}{
  Ein \textbf{Bit} (binary digit) ist eine Informationseinheit,
  die genau einen von zwei Werten annehmen kann: $0$ oder $1$.
}

\ex{Binärdarstellung}{
  Die Zahl $10$ lässt sich im Binärsystem wie folgt darstellen:
  \[
    (1010)_2 = 1\cdot 2^3 + 0\cdot 2^2
             + 1\cdot 2^1 + 0\cdot 2^0 = 10.
  \]
}

\nt{
  Unterschiedliche formale Definitionen von Algorithmen können
  unter geeigneten Voraussetzungen dieselbe Berechenbarkeit
  beschreiben.
}

\nt{
  Ein typischer Kompilierungsprozess lässt sich vereinfacht
  folgendermaßen darstellen:
  \[
    \text{Quellcode}
    \longrightarrow
    \text{Maschinencode}.
  \]
  Dazwischen können weitere Darstellungen liegen, beispielsweise
  Zwischencode oder eine interne Repräsentation.
}

\dfn{Programm}{
  Ein \textbf{Programm} ist eine Zeichenfolge, die den syntaktischen
  Regeln einer Programmiersprache genügt. Seine Ausführung wird
  durch die Semantik der jeweiligen Programmiersprache bestimmt.
}

\section {Primzahlentester}
\dfn{Primzahl} {
  Eine natürliche Zahl $n \in \bbN$ heißt \textbf{prim}, wenn $n \ge 2$ ist und es keine zwei natürliche Zahlen $a$ und $b$ gibt mit $a \ge 2, b \ge 2$, sodass $n = a \cdot b$.
}
\dfn{Primzahltest}{
  Das Entscheidungsproblem des Primzahltests wird definiert als
  \[
    P :=
    \left\{
      (n,e)\in\bbN\times\{\text{ja},\text{nein}\}
      \;\middle|\;
      e=\text{ja}\iff n\text{ ist prim}
    \right\}.
  \]
  Eingabe: \(n\in\bbN\)

  Frage: Ist \(n\) prim?

  Lösung: Probiere alle \(a,b\) mit
  \[
    2\le a,b\le \frac{n}{2}
  \]
  durch, bis \(a\cdot b=n\) gilt oder alle Möglichkeiten geprüft wurden.
}

\section{Landau-Symbole}

\dfn{Landau-Symbole}{
  Sei \(g\colon\bbN\to\bbR^+\). Dann definieren wir:
  \begin{align*}
    O(g)
    &:= \left\{
      f\colon\bbN\to\bbR^+
      \;\middle|\;
      \exists \alpha\in\bbR^+\,
      \exists n_0\in\bbN\,
      \forall n\ge n_0:
      f(n)\le \alpha\cdot g(n)
    \right\},\\
    \Omega(g)
    &:= \left\{
      f\colon\bbN\to\bbR^+
      \;\middle|\;
      \exists \alpha\in\bbR^+\,
      \exists n_0\in\bbN\,
      \forall n\ge n_0:
      f(n)\ge \alpha\cdot g(n)
    \right\},\\
    \Theta(g)
    &:= O(g)\cap\Omega(g).
  \end{align*}
}

\ex{}{
  Sei
  \[
    f(n)=2n^2-n+1.
  \]
  Dann gilt
  \begin{align*}
    f&\in O(n^2)
      &&\text{mit }\alpha=2,\ n_0=1,\\
    f&\in\Omega(n^2)
      &&\text{mit }\alpha=1,\ n_0=1.
  \end{align*}
  Somit folgt
  \[
    f\in\Theta(n^2).
  \]
}

\section{Nicht alles ist berechenbar}

Mit Algorithmen (bzw. C++-Programmen) kann man vieles berechnen,
aber nicht alles. Wir wollen zunächst zeigen, dass jede Sprache,
insbesondere auch die der C++-Programme, abzählbar ist.

Hierzu benötigen wir:

\mlemma{Kodierung von Wörtern}{
  Seien $k,l\in\bbN$. Definiere
  \[
    f_l\colon\{0,\dots,k-1\}^l
    \to\{0,\dots,k^l-1\}
  \]
  durch
  \[
    f_l(w):=\sum_{i=1}^l a_i k^{l-i}
  \]
  für alle $w=a_1\cdots a_l\in\{0,\dots,k-1\}^l$.

  Dann ist $f_l$ wohldefiniert und bijektiv.
}

\begin{myproof}
  Für $w\in\{0,\dots,k-1\}^l$ gilt
  \[
    0\le f_l(w)
    \le\sum_{i=1}^l(k-1)k^{l-i}
    =k^l-1.
  \]
  Somit ist $f_l$ wohldefiniert.

  \medskip
  \textbf{Injektivität:}

  Seien $w,w'\in\{0,\dots,k-1\}^l$ mit
  \[
    w=a_1\cdots a_l,\qquad
    w'=a'_1\cdots a'_l,\qquad w\ne w'.
  \]

  Dann existiert ein kleinster Index $j\in\{1,\dots,l\}$
  mit $a_j\ne a'_j$. Ohne Einschränkung sei $a_j>a'_j$.

  Es gilt
  \[
    f_l(w')
    =\sum_{i=1}^{j-1}a'_i k^{l-i}
    +a'_j k^{l-j}
    +\sum_{i=j+1}^l a'_i k^{l-i}.
  \]

  Wegen
  \begin{align*}
    \sum_{i=1}^{j-1}a'_i k^{l-i}
    &=\sum_{i=1}^{j-1}a_i k^{l-i},\\
    a'_j k^{l-j}
    &\le(a_j-1)k^{l-j},\\
    \sum_{i=j+1}^l a'_i k^{l-i}
    &\le\sum_{i=j+1}^l(k-1)k^{l-i}
     =k^{l-j}-1
  \end{align*}
  folgt
  \begin{align*}
    f_l(w')
    &\le\sum_{i=1}^{j-1}a_i k^{l-i}
      +(a_j-1)k^{l-j}+k^{l-j}-1\\
    &=\sum_{i=1}^{j-1}a_i k^{l-i}
      +a_j k^{l-j}-1\\
    &<f_l(w).
  \end{align*}

  Also gilt $f_l(w')\ne f_l(w)$, und $f_l$ ist injektiv.

  \medskip
  \textbf{Surjektivität:}

  Da
  \[
    |\{0,\dots,k-1\}^l|
    =k^l
    =|\{0,\dots,k^l-1\}|
  \]
  gilt, ist jede injektive Abbildung zwischen diesen
  endlichen Mengen auch surjektiv.

  Somit ist $f_l$ bijektiv.
\end{myproof}


\thm{Abzählbarkeit endlicher Wörter}{
  Sei $A$ eine nichtleere endliche Menge.
  Dann ist die Menge $A^*$ abzählbar.
}

\begin{myproof}
  Sei
  \[
    f\colon A\to\{0,\dots,|A|-1\}
  \]
  eine Bijektion.

  Wir definieren $g\colon A^*\to\bbN$ durch
  \[
    g(a_1\cdots a_l)
    :=1+\sum_{i=0}^{l-1}|A|^i
    +\sum_{i=1}^l f(a_i)|A|^{l-i}.
  \]

  Wir zeigen, dass $g$ injektiv ist.

  Nach dem vorherigen Lemma bildet $g$ die Menge
  der Wörter der Länge $l$ bijektiv auf
  \[
    \left\{
      1+\sum_{i=0}^{l-1}|A|^i,\,
      \dots,\,
      \sum_{i=0}^l|A|^i
    \right\}
  \]
  ab.

  Da diese Mengen für verschiedene Wortlängen
  disjunkt sind, ist $g$ insgesamt injektiv.

  Somit ist $A^*$ abzählbar.
\end{myproof}


\cor{Abzählbarkeit von C++-Programmen}{
  Die Menge aller C++-Programme ist abzählbar.
}

\begin{myproof}
  C++-Programme sind Wörter über einem endlichen Alphabet.

  Nach dem vorherigen Satz ist die Menge aller
  Wörter über diesem Alphabet abzählbar.

  Da Teilmengen abzählbarer Mengen ebenfalls
  abzählbar sind, ist auch die Menge aller
  C++-Programme abzählbar.
\end{myproof}


\thm{Nicht berechenbare Funktionen}{
  Es gibt Funktionen
  \[
    f\colon\bbN\to\{0,1\},
  \]
  die von keinem C++-Programm berechnet werden.
}

\begin{myproof}
  Sei $\mathcal P$ die Menge aller C++-Programme,
  die eine Funktion
  \[
    f\colon\bbN\to\{0,1\}
  \]
  berechnen.

  Nach dem vorherigen Korollar ist $\mathcal P$
  abzählbar. Sei daher
  \[
    g\colon\mathcal P\to\bbN
  \]
  eine injektive Abbildung.

  Für jedes $P\in\mathcal P$ bezeichne $f_P$
  die von $P$ berechnete Funktion.

  Wir definieren eine Funktion
  \[
    f\colon\bbN\to\{0,1\}
  \]
  durch
  \[
    f(g(P)):=1-f_P(g(P))
    \qquad\forall P\in\mathcal P.
  \]

  Für alle übrigen $n\in\bbN$ setzen wir
  beispielsweise $f(n):=0$.

  Da $g$ injektiv ist, ist $f$ wohldefiniert.

  Für jedes $P\in\mathcal P$ gilt
  \[
    f(g(P))=1-f_P(g(P))\ne f_P(g(P)).
  \]

  Somit ist $f\ne f_P$ für alle $P\in\mathcal P$.

  Folglich wird $f$ von keinem C++-Programm berechnet.
\end{myproof}

\section{Das Halteproblem}

Sei $\mathcal Q$ die Menge aller C++-Programme, die eine
natürliche Zahl als Eingabe akzeptieren.

Sei
\[
  g\colon\bbN\to\mathcal Q
\]
eine surjektive Funktion.

Da $\mathcal Q$ nach dem vorherigen Korollar abzählbar ist,
existiert eine injektive Funktion
\[
  f\colon\mathcal Q\to\bbN.
\]

Wir können daher
\[
  g(f(Q)):=Q
  \qquad\forall Q\in\mathcal Q
\]
setzen und für alle $n\notin f(\mathcal Q)$ definieren:
\[
  g(n):=Q_0
\]
für ein beliebiges, fest gewähltes $Q_0\in\mathcal Q$.

\dfn{Halteproblem}{
  Wir definieren die \textbf{Haltefunktion}
  \[
    h\colon\bbN\times\bbN\to\{0,1\}
  \]
  durch
  \[
    h(x,y):=
    \begin{cases}
      1, & \text{falls }g(x)\text{ bei Eingabe von }y
           \text{ nach endlich vielen Schritten hält},\\
      0, & \text{sonst}.
    \end{cases}
  \]

  Das zugehörige Entscheidungsproblem heißt
  \textbf{Halteproblem}.
}

\thm{Unentscheidbarkeit des Halteproblems}{
  Die Haltefunktion $h$ wird von keinem
  C++-Programm berechnet.
}

\begin{myproof}
  Wir führen einen \textbf{Widerspruchsbeweis}.

  Angenommen, es existiert ein C++-Programm $P$,
  das die Haltefunktion $h$ berechnet.

  Dann können wir ein Programm $Q$ konstruieren,
  das bei Eingabe von $x\in\bbN$ zunächst
  $h(x,x)$ berechnet und anschließend:
  \begin{itemize}
    \item terminiert, falls $h(x,x)=0$,
    \item in einer Endlosschleife läuft, falls $h(x,x)=1$.
  \end{itemize}

  Da $Q\in\mathcal Q$ und $g$ surjektiv ist,
  existiert ein $q\in\bbN$ mit
  \[
    g(q)=Q.
  \]

  Nach Definition der Haltefunktion gilt
  \[
    h(q,q)=1
    \iff
    g(q)\text{ terminiert bei Eingabe von }q.
  \]

  Wegen $g(q)=Q$ folgt
  \[
    h(q,q)=1
    \iff
    Q\text{ terminiert bei Eingabe von }q.
  \]

  Nach der Konstruktion von $Q$ gilt jedoch
  \[
    Q\text{ terminiert bei Eingabe von }q
    \iff h(q,q)=0.
  \]

  Zusammen erhalten wir
  \[
    \boxed{h(q,q)=1\iff h(q,q)=0,}
  \]
  ein Widerspruch.

  Somit kann kein C++-Programm die Haltefunktion
  $h$ berechnen.
\end{myproof}

\nt{
  Der Satz gilt ebenso für jede andere
  hinreichend mächtige Programmiersprache und
  allgemein für Algorithmen.

  Man sagt daher:
  \[
    \boxed{\text{Das Halteproblem ist nicht entscheidbar.}}
  \]
}
